#include <tunables/global>

# AppArmor profile for Bifrost.
# Based on Docker's default container profile (what the app runs with anyway),
# plus restrictions Bifrost never needs: no raw sockets, no mounts, no writes to its own code,
# no access to host secrets. The Supervisor renames the profile to the installed slug.
profile bifrost flags=(attach_disconnected,mediate_deleted) {
  # s6-overlay (init) and Python need ordinary file access, TCP/UDP to Home Assistant and signals
  file,
  capability,
  network inet,
  network inet6,
  network unix,
  deny network raw,
  deny network packet,
  umount,

  signal (send,receive) peer=@{profile_name},
  signal (receive) peer=unconfined,
  ptrace (trace,read,tracedby,readby) peer=@{profile_name},

  # the app code is read-only at runtime
  deny /opt/bifrost/** wl,
  deny /run.sh wl,

  # no mounts, no kernel or firmware tinkering (as in Docker's default profile)
  deny mount,
  deny pivot_root,
  deny @{PROC}/* w,
  deny @{PROC}/{[^1-9],[^1-9][^0-9],[^1-9s][^0-9y][^0-9s],[^1-9][^0-9][^0-9][^0-9/]*}/** w,
  deny @{PROC}/sys/[^k]** w,
  deny @{PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/kcore rwklx,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/kernel/security/** rwklx,

  # Bifrost never needs SSH keys or other host secrets
  deny /root/.ssh/** rwklx,
  deny /ssl/** rwklx,
  deny /share/** wklx,
}
